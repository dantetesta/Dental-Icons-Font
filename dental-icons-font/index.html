styles.css
@font-face {
  font-family: "Dental Icons Up";
  src: url("DentalIconsUpper-Regular.woff2") format("woff2");
  font-style: normal;
  font-weight: 400;
  font-display: swap;
}

@font-face {
  font-family: "Dental Icons Down";
  src: url("DentalIconsLower-Regular.woff2") format("woff2");
  font-style: normal;
  font-weight: 400;
  font-display: swap;
}

.arcada-superior { font-family: "Dental Icons Up"; }
.arcada-inferior { font-family: "Dental Icons Down"; }
.arcada-superior, .arcada-inferior {
  font-synthesis: none;
  font-feature-settings: "ccmp" 1, "liga" 1, "calt" 1, "kern" 1;
}
